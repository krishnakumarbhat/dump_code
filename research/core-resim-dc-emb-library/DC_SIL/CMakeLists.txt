# Without this, both SRR and MRR would share the same dgps_iface and protobuf state at runtime.

	add_custom_command(TARGET ${PROJECTNAME}
		POST_BUILD
		COMMAND ${CMAKE_COMMAND} -E copy_if_different
		"${DGPS_LIB_DIR}/libSensorModel2.so"
		"$<TARGET_FILE_DIR:${PROJECTNAME}>/libSensorModel2.so"
		COMMAND ${CMAKE_COMMAND} -E copy_if_different
		"${DGPS_LIB_DIR}/libSMValidation.so"
		"$<TARGET_FILE_DIR:${PROJECTNAME}>/libSMValidation.so"
